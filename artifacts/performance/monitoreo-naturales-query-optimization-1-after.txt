			       VIDAD_ECoNOMICA, P.SECTOR SECTOR_ID, S.DESCRIPCION SECTOR,

V_SOCIOS_REPRESENTANTES        SELECT 'SOCIO' TIPO_CONDICION_ACTUA_DESC,
					 VS.SOCIO_ID ID_PERSONA,
					 VS.NOMBRE_SOCIO NOMBRES_PERSONA,
					 VS.NUMERO_IDENTIFICACION,
					 VS.RTN,
					 VS.PORCENTAJE_PARTICIPACION PORCENTAJE_PARTICIPACION,
					 VS.NOMBRE_PATRONO NOMBRE_EMPRESA,
					 VS.NUMERO_PATRONAL,
					 A.DESCRIPCION ACTIVIDA_ECONOMICA,
					 (SELECT DESCRIPCION
					    FROM HONDURAS
					   WHERE DEPARTAMENTO = P.DEPARTAMENTO AND MUNICIPIO = '00')
					    DEPARTAMENTO,
					 (SELECT DESCRIPCION
					    FROM HONDURAS
					   WHERE     DEPARTAMENTO = P.DEPARTAMENTO
						 AND MUNICIPIO = P.MUNICIPIO
						 AND ALDEA = '00')
					    MUNICIPIO,
					 P.DIREVIEJA DIRECCION,
					 L.DESCRIPCION REGIONAL,
					 TRUNC (VS.FECHA_REGISTRO) FECHA_REGISTRO,
					 VS.DATOS_EMPRESA_ID,
					 VS.ES_PEP,
					 TF.TIPO_FUNCIONARIO_DESCRIPCION,
					 VS.ESTADO_REGISTRO,
					 US0.NOMBRE_USUARIO USUARIO_CREO,
					 VS.FECHA_CREACION FECHA_CREO,
					 US1.NOMBRE_USUARIO USUARIO_MODIFICO,
					 VS.FECHA_MODIFICO,
					 vs.PERIODO_PEP_INICIO,
					 vs.PERIODO_PEP_FIN
				    FROM V_SOCIOS VS
					 LEFT JOIN PATRONOS P ON VS.NUMERO_PATRONAL = P.NUMEPATRO
					 LEFT JOIN ACTIECON A ON P.ACTIECON = A.ACTIECON
					 LEFT JOIN LUGARES L ON P.LUGAR = L.LUGAR
					 LEFT JOIN TIPO_FUNCIONARIO TF
					    ON VS.TIPO_FUNCIONARIO_ID = TF.TIPO_FUNCIONARIO_ID
					 LEFT JOIN USUARIO US0 ON US0.ID_USUARIO = VS.USUARIO_CREO
					 LEFT JOIN USUARIO US1 ON US1.ID_USUARIO = VS.USUARIO_MODIFICO
				  UNION
				  SELECT VR.TIPO_CONDICION_ACTUA_DESC,
					 VR.REPRESENTANTE_ID ID_PERSONA,
					 VR.NOMBRES_PERSONA,
					 VR.NUMERO_IDENTIFICACION,
